    local dom = require "luaxml-domobject"
    f = io.open("../xml/research.xml", 'r')
    xmlfile = f:read('*all')
    f:close()

    local obj = dom.parse(xmlfile)
    
    months = {january = 1, february = 2, march = 03, april = 4, may = 5, june = 6, july = 7, august = 8, september = 9, october = 10, november = 11, december = 12}

    -- Function for parsing a note with HTML formatting

    function parse_note(str)
      local ps = str:gsub("<notes>(.*)</notes>", "%1")
      if ps:match("<%!%[CDATA%[") ~= nil then 
        ps = ps:gsub("\n", "")
        ps = ps:gsub("\r", "")
        ps = ps:gsub("%s%s+", " ")
        ps = ps:gsub("<%!%[CDATA%[", "")
        ps = ps:gsub("%]%]>", "")
        ps = ps:gsub("<span style=\"font%-weight: bold;\">", "\\textbf{")
        ps = ps:gsub("<span style=\"font%-style:italic;\">", "\\emph{")
        ps = ps:gsub("</span>", "}")
        ps = ps:gsub("<strong>", "\\textbf{")
        ps = ps:gsub("<em>", "\\emph{")
        ps = ps:gsub("</strong>", "}")
        ps = ps:gsub("</em>", "}")
        ps = ps:gsub("{%s+", "{")
        ps = ps:gsub("%s+}", "}")
        ps = ps:gsub("%s\"", " ``")
        ps = ps:gsub("(%a)\"", "%1''")
        ps = ps:gsub("%s+$", "")
      end
      return ps
    end

    -- Helper: append a period only if the text does not already end with one
    -- (ignores trailing closing braces, quotes and whitespace from the formatting)

    function add_period(str)
      if str:gsub("[}%s'`]*$", ""):sub(-1) == "." then
        return str
      else
        return str .. "."
      end
    end

    -- Function for printing publications (econ and other)

    function publications(pub_type)
      year = "a"
      -- sort the entries
      tab = obj:query_selector(pub_type)
      table.sort(tab, function(a, b)
          for _, el in ipairs(a:get_children()) do
            if el:is_element() then
              el_name = el:get_element_name()
              el_text = el:get_text()
              if el_name == "date" then
                year_sort = string.match(el_text, "%d%d%d%d")
                for word in el_text:gmatch("%w+") do
                  if months[word:lower()] ~= nil then month_sort = months[word:lower()] end
                end
                dateA = year_sort + month_sort / 12
              end
            end
          end
          for _, el in ipairs(b:get_children()) do
            if el:is_element() then
              el_name = el:get_element_name()
              el_text = el:get_text()
              if el_name == "date" then
                year_sort = string.match(el_text, "%d%d%d%d")
                for word in el_text:gmatch("%w+") do
                  if months[word:lower()] ~= nil then month_sort = months[word:lower()] end
                end
                dateB = year_sort + month_sort / 12
              end
            end
          end
          return dateA > dateB
        end
      )
      for _, wp in ipairs(tab) do
        wp_text = ""
        for _, el in ipairs(wp:get_children()) do
          if el:is_element() then
            el_text = el:get_text()
            el_text = el_text:gsub("&", "\\&")
            el_name = el:get_element_name()

            if el_name == "date" then
              if string.match(el_text, "%d%d%d%d") ~= year 
              then 
                if string.match(el_text, "%d+") == nil 
                  then year = el_text 
                  else year = string.match(el_text, "%d+") 
                end
                wp_text = year .. " & " .. wp_text
              else 
                wp_text = "&" .. wp_text
              end
            elseif el_name == "title" then wp_text = wp_text .. "``" .. el_text
            elseif el_name == "coauthors" then
              if el_text ~= "" 
                then wp_text = wp_text .. "'' (with " .. el_text .. "), " 
                else wp_text = wp_text .. ",'' "
              end
              elseif el_name == "journal" and el_text ~= "" then wp_text = wp_text .. "\\textbf{\\emph{" .. el_text .. "}}"
              elseif el_name == "vol_issue" and el_text ~= "" then wp_text = wp_text .. " " .. el_text .. ": "
              elseif el_name == "pages" and el_text ~= "" then wp_text = wp_text .. el_text:gsub("(%d+)-(%d+)", "%1--%2")
              elseif el_name == "notes" and el_text ~= "" then
                el_text = obj:serialize(el)
                el_text = parse_note(el_text)
                wp_text = wp_text .. ". {\\small " .. el_text .. "}"
                wp_text = wp_text:gsub("''%.", ".''")
            end
          end
        end
        print(wp_text)
        tex.print(add_period(wp_text) .. " \\\\")
      end
    end

    -- Function for printing working papers/work in progress (econ and other)

    function work(work_type)
      -- sort the entries
      tab = obj:query_selector(work_type)
      if work_type == "wp" then
        table.sort(tab, function(a, b)
            for _, el in ipairs(a:get_children()) do
              if el:is_element() then
                el_name = el:get_element_name()
                el_text = el:get_text()
                if el_name == "date" then
                  year_sort = string.match(el_text, "%d%d%d%d")
                  for word in el_text:gmatch("%w+") do
                    if months[word:lower()] ~= nil then month_sort = months[word:lower()] end
                  end
                  dateA = year_sort + month_sort / 12
                end
              end
            end
            for _, el in ipairs(b:get_children()) do
              if el:is_element() then
                el_name = el:get_element_name()
                el_text = el:get_text()
                if el_name == "date" then
                  year_sort = string.match(el_text, "%d%d%d%d")
                  for word in el_text:gmatch("%w+") do
                    if months[word:lower()] ~= nil then month_sort = months[word:lower()] end
                  end
                  dateB = year_sort + month_sort / 12
                end
              end
            end
            return dateA > dateB
          end
        )
      end
      for _, wp in ipairs(tab) do
        wp_text = ""
        for _, el in ipairs(wp:get_children()) do
          if el:is_element() then
            el_text = el:get_text()
            el_name = el:get_element_name()

            if el_name == "date" then
              year = string.match(el_text, "%d%d%d%d")
              if year == nil then wp_text = "\\hskip 1em & " .. wp_text
              else 
                for word in el_text:gmatch("%w+") do
                  if months[word:lower()] ~= nil then month = months[word:lower()] end
                end
                if month < 10 then wp_text = year .. "/0" .. month .. " & " .. wp_text
                else wp_text = year .. "/" .. month .. " & " .. wp_text 
                end
              end
            elseif el_name == "title" then wp_text = wp_text .. "``" .. el_text
            elseif el_name == "coauthors" then
              if el_text ~= "" 
                then wp_text = wp_text .. "'' (with " .. el_text .. ")." 
                else wp_text = wp_text .. ".''"
              end
              elseif el_name == "notes" and el_text ~= "" then 
                el_text = obj:serialize(el)
                el_text = parse_note(el_text)
                wp_text = wp_text .. " " .. add_period(el_text)
                wp_text = wp_text:gsub("''%.", ".''")
            end
          end
        end
        print(wp_text)
        tex.print(wp_text .. "\\\\")
      end
    end


  \end{luacode*}
